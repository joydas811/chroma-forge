    <script>
    function generateSignal() {
    const pair = document.getElementById('assetPair').value;
    const tf = document.getElementById('timeFrame').value;

    const isBuy = Math.random() >= 0.5;
    const strength = Math.floor(Math.random() * (98 - 80 + 1)) + 80;

    const now = new Date();
    const timeString = String(now.getHours()).padStart(2, '0') + ':' + String(now.getMinutes()).padStart(2, '0');

    const card = document.getElementById('outputCard');
    const dirHeader = document.getElementById('dirHeader');
    const actionTitle = document.getElementById('actionTitle');

    if (isBuy) {
    dirHeader.innerHTML = "GO FOR UP ▲";
    dirHeader.style.color = "#00ffcc";
    actionTitle.innerHTML = "BUY";
    actionTitle.style.color = "#00ffcc";
    actionTitle.style.textShadow = "0 0 10px rgba(0, 255, 204, 0.5)";
    } else {
    dirHeader.innerHTML = "GO FOR DOWN ▼";
    dirHeader.style.color = "#ff5566";
    actionTitle.innerHTML = "SELL";
    actionTitle.style.color = "#ff3344";
    actionTitle.style.textShadow = "0 0 10px rgba(255, 51, 68, 0.5)";
    }

    document.getElementById('resPair').innerText = pair;
    document.getElementById('resTime').innerText = timeString;
    document.getElementById('resTf').innerText = tf;
    document.getElementById('resStrength').innerText = strength + "% LOCK STRENGTH";

    card.style.display = "block";
    }

    function copySignal() {
    const action = document.getElementById('actionTitle').innerText;
    const pair = document.getElementById('resPair').innerText;
    const time = document.getElementById('resTime').innerText;
    const tf = document.getElementById('resTf').innerText;
    const strength = document.getElementById('resStrength').innerText;

    const text = `🚀 *CHROMA FORGE SIGNAL* 🚀\n\n` +
    `📉 *DIRECTION: ${action}*\n` +
    `🔹 *Pair:* ${pair}\n` +
    `⏰ *Time:* ${time}\n` +
    `⏱ *TF:* ${tf}\n` +
    `🔥 *Lock Strength:* ${strength}\n\n` +
    `_Channel: Trading By MinhaZ_`;

    navigator.clipboard.writeText(text);
    alert("Signal copied to clipboard!");
    }
